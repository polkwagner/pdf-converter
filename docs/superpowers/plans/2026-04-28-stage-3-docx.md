**Goal:** Add DOCX→markdown conversion to `convert.py`. Default behavior is **lean** (per spec §4.6 reversed): comments dropped, tracked changes accepted-final, images dropped (placeholder kept), but **footnotes always preserved** as `[^N]` markdown footnotes with a `## Footnotes` section. Three opt-in flags: `--full` adds a `## Reviewer Comments` appendix, `--show-revisions` renders ins/del inline, `--keep-images` extracts images to disk.

**Architecture:** New `materials/formats/docx.py`. Docling does the prose/heading/table conversion; python-docx (mandatory in this stage) handles comments, footnotes, and tracked changes via lxml on the document's XML tree. Section markers reuse the §5.2 sanitize_heading_text + numbered-marker pattern from HTML.

**Tech Stack:** Same as Stage 2 plus `python-docx>=1.1.0` (mandatory).

**Spec:** `docs/superpowers/specs/2026-04-28-materials-converter-design.md` §4.6 (DOCX defaults), §5.3 (DOCX pipeline + comment-extraction scope), §6.1 (cheap checks), §8 stage 3.

**Scope guardrail:** comment-extraction is **top-level comments only** (`comments.xml`). Threaded replies in `commentsExtended.xml` are a documented limitation, deferred to a follow-up after stage 5.

```python
"""Build tests/fixtures/sample_with_comments.docx — a Word document that exercises:
  - H1 + H2 headings (for section markers)
  - One footnote (preserved by default per spec §4.6)
  - One Word comment (dropped by default; appears with --full)
  - One tracked-changes pair (insertion + deletion; accepted-final by default)

python-docx builds the prose and headings. Comments are injected via lxml
since python-docx's high-level API doesn't support them. Tracked changes
likewise go in via lxml.

Run from repo root:
    ./venv/bin/python tests/fixtures/build/build_docx.py
"""
from __future__ import annotations

import zipfile
from pathlib import Path

from docx import Document
from docx.oxml.ns import qn
from lxml import etree

OUT = Path(__file__).resolve().parent.parent / "sample_with_comments.docx"

# Word's namespace prefixes (qn() resolves these to fully-qualified names).
W = "http://schemas.openxmlformats.org/wordprocessingml/2006/main"
NSMAP = {"w": W}


def _build_base_doc() -> Document:
    """Build the prose with python-docx — headings, paragraphs, a footnote-style
    reference. We'll inject comments and revisions via lxml after."""
    doc = Document()
    doc.add_heading("Memo Title", level=1)

    p1 = doc.add_paragraph(
        "This is the opening paragraph. It contains a phrase that a "
    )
    # The next run is the comment-anchor target — we'll inject commentRange
    # markers around it in step 3.
    run_anchor = p1.add_run("contested claim")
    p1.add_run(" worth annotating, plus a footnote reference.")

    doc.add_heading("Discussion", level=2)
    doc.add_paragraph(
        "This paragraph contains both a footnote reference and a tracked "
        "change. The footnote text appears at the end of the document."
    )

    # Footnote: python-docx doesn't add real Word footnotes via its high-level
    # API, but Docling handles native Word footnotes. We add a paragraph that
    # mimics a footnote-style reference; the actual footnote XML is injected
    # in step 4. (Docling reads the underlying XML directly, not python-docx's
    # object model.)

    return doc


def _save_via_python_docx(doc: Document) -> None:
    doc.save(str(OUT))


def _inject_comment_into_archive() -> None:
    """Read the .docx zip, add a comments.xml part, modify document.xml to
    include commentRangeStart/End around 'contested claim', and patch the
    relationship table. Returns nothing — modifies OUT in place via repack.
    """
    # Read all parts from the zip
    parts: dict[str, bytes] = {}
    with zipfile.ZipFile(OUT, "r") as zin:
        for name in zin.namelist():
            parts[name] = zin.read(name)

    # 1. Build comments.xml
    comments_xml = (
        b'<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n'
        b'<w:comments xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main">'
        b'<w:comment w:id="0" w:author="Sarah Chen" w:date="2025-09-12T10:00:00Z" w:initials="SC">'
        b'<w:p><w:r><w:t>Consider citing Bilski here.</w:t></w:r></w:p>'
        b'</w:comment>'
        b'</w:comments>'
    )
    parts["word/comments.xml"] = comments_xml

    # 2. Patch document.xml — wrap "contested claim" with commentRangeStart/End +
    #    add a commentReference run after.
    doc_tree = etree.fromstring(parts["word/document.xml"])
    body = doc_tree.find(qn("w:body"))
    for paragraph in body.findall(qn("w:p")):
        for run in paragraph.findall(qn("w:r")):
            t = run.find(qn("w:t"))
            if t is not None and t.text == "contested claim":
                # Insert commentRangeStart before the run, End + Reference after.
                idx = list(paragraph).index(run)
                start = etree.SubElement(paragraph, qn("w:commentRangeStart"))
                start.set(qn("w:id"), "0")
                paragraph.remove(start)
                paragraph.insert(idx, start)

                end = etree.SubElement(paragraph, qn("w:commentRangeEnd"))
                end.set(qn("w:id"), "0")
                paragraph.remove(end)
                paragraph.insert(idx + 2, end)

                ref_run = etree.SubElement(paragraph, qn("w:r"))
                ref = etree.SubElement(ref_run, qn("w:commentReference"))
                ref.set(qn("w:id"), "0")
                paragraph.remove(ref_run)
                paragraph.insert(idx + 3, ref_run)
                break
        else:
            continue
        break
    parts["word/document.xml"] = etree.tostring(
        doc_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    # 3. Patch the document's relationship file so it references comments.xml.
    rels_path = "word/_rels/document.xml.rels"
    rels_tree = etree.fromstring(parts[rels_path])
    R = "http://schemas.openxmlformats.org/package/2006/relationships"
    new_rel = etree.SubElement(rels_tree, f"{{{R}}}Relationship")
    new_rel.set("Id", "rIdComments")
    new_rel.set(
        "Type",
        "http://schemas.openxmlformats.org/officeDocument/2006/relationships/comments",
    )
    new_rel.set("Target", "comments.xml")
    parts[rels_path] = etree.tostring(
        rels_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    # 4. Patch [Content_Types].xml to declare the comments part.
    ct_path = "[Content_Types].xml"
    ct_tree = etree.fromstring(parts[ct_path])
    CT = "http://schemas.openxmlformats.org/package/2006/content-types"
    override = etree.SubElement(ct_tree, f"{{{CT}}}Override")
    override.set("PartName", "/word/comments.xml")
    override.set(
        "ContentType",
        "application/vnd.openxmlformats-officedocument.wordprocessingml.comments+xml",
    )
    parts[ct_path] = etree.tostring(
        ct_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    # 5. Repack.
    with zipfile.ZipFile(OUT, "w", zipfile.ZIP_DEFLATED) as zout:
        for name, data in parts.items():
            zout.writestr(name, data)


def _inject_tracked_change() -> None:
    """Inject a w:ins (insertion) and a w:del (deletion) into the second
    body paragraph so --show-revisions has something to render."""
    parts: dict[str, bytes] = {}
    with zipfile.ZipFile(OUT, "r") as zin:
        for name in zin.namelist():
            parts[name] = zin.read(name)

    doc_tree = etree.fromstring(parts["word/document.xml"])
    body = doc_tree.find(qn("w:body"))

    # Find the "Discussion" heading paragraph, then the next paragraph after it.
    paragraphs = body.findall(qn("w:p"))
    target = None
    for i, p in enumerate(paragraphs):
        for r in p.findall(qn("w:r")):
            t = r.find(qn("w:t"))
            if t is not None and t.text and "tracked change" in t.text:
                target = p
                break
        if target is not None:
            break

    if target is None:
        return  # silently skip — fixture still useful without this

    # Append " inserted-text" via w:ins (Word's "track changes" insertion).
    ins = etree.SubElement(target, qn("w:ins"))
    ins.set(qn("w:id"), "100")
    ins.set(qn("w:author"), "Sarah Chen")
    ins.set(qn("w:date"), "2025-09-12T10:00:00Z")
    ins_run = etree.SubElement(ins, qn("w:r"))
    ins_t = etree.SubElement(ins_run, qn("w:t"))
    ins_t.text = " ADDED-TEXT"
    ins_t.set(qn("xml:space"), "preserve")

    # Append a w:del element with a deleted text.
    delete = etree.SubElement(target, qn("w:del"))
    delete.set(qn("w:id"), "101")
    delete.set(qn("w:author"), "Sarah Chen")
    delete.set(qn("w:date"), "2025-09-12T10:00:00Z")
    del_run = etree.SubElement(delete, qn("w:r"))
    del_t = etree.SubElement(del_run, qn("w:delText"))
    del_t.text = " REMOVED-TEXT"
    del_t.set(qn("xml:space"), "preserve")

    parts["word/document.xml"] = etree.tostring(
        doc_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    with zipfile.ZipFile(OUT, "w", zipfile.ZIP_DEFLATED) as zout:
        for name, data in parts.items():
            zout.writestr(name, data)


def _inject_footnote() -> None:
    """Inject a footnote part + a footnoteReference in the body so Docling
    surfaces it as native Word footnote content."""
    parts: dict[str, bytes] = {}
    with zipfile.ZipFile(OUT, "r") as zin:
        for name in zin.namelist():
            parts[name] = zin.read(name)

    # 1. footnotes.xml — Word reserves IDs -1 (separator), 0 (continuation), and 1+.
    footnotes_xml = (
        b'<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n'
        b'<w:footnotes xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main">'
        b'<w:footnote w:type="separator" w:id="-1"><w:p><w:r><w:separator/></w:r></w:p></w:footnote>'
        b'<w:footnote w:type="continuationSeparator" w:id="0"><w:p><w:r><w:continuationSeparator/></w:r></w:p></w:footnote>'
        b'<w:footnote w:id="1">'
        b'<w:p><w:r><w:t>This is the footnote text. It contains a citation: Bilski v. Kappos, 561 U.S. 593 (2010).</w:t></w:r></w:p>'
        b'</w:footnote>'
        b'</w:footnotes>'
    )
    parts["word/footnotes.xml"] = footnotes_xml

    # 2. Add a footnoteReference run at the end of the second body paragraph.
    doc_tree = etree.fromstring(parts["word/document.xml"])
    body = doc_tree.find(qn("w:body"))
    paragraphs = body.findall(qn("w:p"))
    # Append to the last paragraph (the Discussion paragraph).
    target = None
    for p in paragraphs:
        for r in p.findall(qn("w:r")):
            t = r.find(qn("w:t"))
            if t is not None and t.text and "footnote text appears" in t.text:
                target = p
                break
        if target is not None:
            break
    if target is None and paragraphs:
        target = paragraphs[-1]

    if target is not None:
        ref_run = etree.SubElement(target, qn("w:r"))
        ref = etree.SubElement(ref_run, qn("w:footnoteReference"))
        ref.set(qn("w:id"), "1")

    parts["word/document.xml"] = etree.tostring(
        doc_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    # 3. Add the footnotes relationship.
    rels_path = "word/_rels/document.xml.rels"
    rels_tree = etree.fromstring(parts[rels_path])
    R = "http://schemas.openxmlformats.org/package/2006/relationships"
    new_rel = etree.SubElement(rels_tree, f"{{{R}}}Relationship")
    new_rel.set("Id", "rIdFootnotes")
    new_rel.set(
        "Type",
        "http://schemas.openxmlformats.org/officeDocument/2006/relationships/footnotes",
    )
    new_rel.set("Target", "footnotes.xml")
    parts[rels_path] = etree.tostring(
        rels_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    # 4. Add Content_Types entry for footnotes.xml.
    ct_path = "[Content_Types].xml"
    ct_tree = etree.fromstring(parts[ct_path])
    CT = "http://schemas.openxmlformats.org/package/2006/content-types"
    override = etree.SubElement(ct_tree, f"{{{CT}}}Override")
    override.set("PartName", "/word/footnotes.xml")
    override.set(
        "ContentType",
        "application/vnd.openxmlformats-officedocument.wordprocessingml.footnotes+xml",
    )
    parts[ct_path] = etree.tostring(
        ct_tree, xml_declaration=True, encoding="UTF-8", standalone=True,
    )

    with zipfile.ZipFile(OUT, "w", zipfile.ZIP_DEFLATED) as zout:
        for name, data in parts.items():
            zout.writestr(name, data)


def build() -> Path:
    doc = _build_base_doc()
    _save_via_python_docx(doc)
    _inject_comment_into_archive()
    _inject_tracked_change()
    _inject_footnote()
    return OUT


if __name__ == "__main__":
    p = build()
    size_kb = p.stat().st_size / 1024
    print(f"Wrote {p} ({size_kb:.1f} KB)")
```

## Task 4: Create `materials/formats/docx.py` with DOCXConverter

**Files:**
- Create: `materials/formats/docx.py`

The big task. Stages of the converter:
1. Read the .docx with python-docx → extract comments, footnotes, revisions from XML
2. Pass file to Docling for headings/paragraphs/tables markdown
3. Post-process markdown:
   - Insert `<!-- Section K: heading -->` markers at H1/H2
   - If `--show-revisions`: walk the body XML, find `w:ins`/`w:del`, locate matching text in markdown, wrap in `[+ ... +]` / `[- ... -]`
   - If lean default (no --show-revisions): the markdown already reflects accepted-final because Docling renders runs not wrapped in `w:del` — verify this on the fixture
   - Footnotes: append `## Footnotes` section with the footnote text. Replace any in-prose footnote markers with `[^N]` references
   - If `--full`: append `## Reviewer Comments` with author, date, comment text, and the referenced phrase quoted (best-effort inline anchor `[C1]` if the referenced text is found in the markdown)
   - If `--keep-images`: extract images to disk and reference via `![](output_files/img-N.png)`. If not, replace embedded image references with `<!-- image -->`
4. Cheap verifier: non-empty + word retention ≥90%
5. Statistics with canonical schema
6. Write output

- [ ] **Step 1: Write `materials/formats/docx.py`**

```python
"""DOCXConverter — convert Word DOCX files to markdown using Docling +
python-docx.

Stage 3: lean default per spec §4.6 reversed — comments dropped, revisions
accepted-final, images dropped (placeholder kept). Footnotes always preserved.
Three opt-in flags: --full (comments appendix), --show-revisions (inline
ins/del markers), --keep-images (extract images to disk).

Comment-extraction scope: top-level comments only (`comments.xml`).
Threaded replies in `commentsExtended.xml` are a known limitation deferred
to a follow-up after stage 5.
"""
from __future__ import annotations

import logging
import re
from dataclasses import dataclass
from pathlib import Path

from docling.document_converter import DocumentConverter

from materials.core.base import BaseConverter, ConversionOptions, ConversionResult
from materials.core.output import default_output_path, sanitize_heading_text
from materials.core.verify import (
    VerifyReport,
    check_non_empty,
    check_word_retention,
    count_words,
)


DOCX_WORD_RETENTION_MIN = 0.90

_HEADING_RE = re.compile(r"^(#{1,2})\s+(.+?)\s*$", re.MULTILINE)

# Word XML namespace
_W_NS = "http://schemas.openxmlformats.org/wordprocessingml/2006/main"


@dataclass
class _CommentEntry:
    id: str
    author: str
    date: str
    text: str
    referenced_text: str  # the body-text snippet the comment is anchored to


@dataclass
class _Revision:
    kind: str  # "ins" or "del"
    text: str


def _qn(tag: str) -> str:
    """Resolve a w:foo tag to its fully-qualified Clark notation."""
    return f"{{{_W_NS}}}{tag}"


def _extract_comments(doc) -> List[_CommentEntry]:
    """Read comments.xml from the .docx archive (via python-docx's part lookup)
    and pair each comment with the body-text snippet it anchors to.

    Best-effort. If anchor positions can't be resolved, the comment still
    appears in the appendix without a referenced-text snippet.
    """
    from docx.opc.constants import RELATIONSHIP_TYPE as RT

    # Find the comments part by relationship type
    comments_part = None
    for rel in doc.part.rels.values():
        if rel.reltype == RT.COMMENTS:
            comments_part = rel.target_part
            break
    if comments_part is None:
        return []

    # Parse comments.xml
    from lxml import etree
    comments_root = etree.fromstring(comments_part.blob)

    # Walk document.xml body to map comment IDs to anchored text snippets
    anchored: dict[str, str] = {}
    body = doc.element.body
    for paragraph in body.findall(_qn("w:p")):
        in_range: dict[str, list[str]] = {}
        for child in paragraph.iter():
            tag = child.tag
            if tag == _qn("w:commentRangeStart"):
                cid = child.get(_qn("w:id"))
                if cid is not None:
                    in_range[cid] = []
            elif tag == _qn("w:commentRangeEnd"):
                cid = child.get(_qn("w:id"))
                if cid is not None and cid in in_range:
                    anchored[cid] = "".join(in_range.pop(cid)).strip()
            elif tag == _qn("w:t") and in_range:
                # Append this run's text to every open commentRange
                for collected in in_range.values():
                    if child.text:
                        collected.append(child.text)
        # If a commentRangeEnd was missing, drop unfinished collectors silently.

    comments: List[_CommentEntry] = []
    for c in comments_root.findall(_qn("w:comment")):
        cid = c.get(_qn("w:id")) or ""
        author = c.get(_qn("w:author")) or "Unknown"
        date = c.get(_qn("w:date")) or ""
        text_parts = []
        for t in c.iter(_qn("w:t")):
            if t.text:
                text_parts.append(t.text)
        comments.append(_CommentEntry(
            id=cid,
            author=author,
            date=date.split("T")[0] if "T" in date else date,
            text="".join(text_parts).strip(),
            referenced_text=anchored.get(cid, ""),
        ))
    return comments


def _extract_footnotes(doc) -> List[tuple[str, str]]:
    """Return [(footnote_id, footnote_text), ...] from footnotes.xml,
    excluding the Word-reserved separator/continuation footnotes (id=-1, 0)."""
    from docx.opc.constants import RELATIONSHIP_TYPE as RT

    footnotes_part = None
    for rel in doc.part.rels.values():
        if rel.reltype == RT.FOOTNOTES:
            footnotes_part = rel.target_part
            break
    if footnotes_part is None:
        return []

    from lxml import etree
    root = etree.fromstring(footnotes_part.blob)
    out: List[tuple[str, str]] = []
    for fn in root.findall(_qn("w:footnote")):
        fid = fn.get(_qn("w:id")) or ""
        if fid in ("-1", "0"):
            continue  # separator / continuation
        text_parts = [t.text for t in fn.iter(_qn("w:t")) if t.text]
        out.append((fid, "".join(text_parts).strip()))
    return out


def _extract_revisions(doc) -> List[_Revision]:
    """Return all w:ins / w:del text content in document order. Used by
    --show-revisions."""
    revisions: List[_Revision] = []
    body = doc.element.body
    for ins in body.iter(_qn("w:ins")):
        text_parts = [t.text for t in ins.iter(_qn("w:t")) if t.text]
        if text_parts:
            revisions.append(_Revision("ins", "".join(text_parts)))
    for delete in body.iter(_qn("w:del")):
        text_parts = [t.text for t in delete.iter(_qn("w:delText")) if t.text]
        if text_parts:
            revisions.append(_Revision("del", "".join(text_parts)))
    return revisions


def _insert_section_markers(markdown: str) -> str:
    """Same logic as html.py's _insert_section_markers — H1/H2 → numbered
    `<!-- Section K: heading -->`. Could be promoted to core/output.py once
    DOCX confirms the pattern, but per Rex's prematureness rule (Stage 1
    review), we wait for a third caller before extracting."""
    matches = list(_HEADING_RE.finditer(markdown))
    if not matches:
        return f"<!-- Section 1: (untitled) -->\n\n{markdown}"
    parts = []
    cursor = 0
    for idx, match in enumerate(matches, start=1):
        parts.append(markdown[cursor:match.start()])
        heading_text = sanitize_heading_text(match.group(2))
        parts.append(f"<!-- Section {idx}: {heading_text} -->\n\n")
        parts.append(match.group(0))
        cursor = match.end()
    parts.append(markdown[cursor:])
    return "".join(parts)


def _render_revisions_inline(markdown: str, revisions: List[_Revision]) -> str:
    """Wrap inserted/deleted text in markers. Matches each revision's text
    in the markdown and wraps it. Insertions appear in the markdown by
    default (Docling renders w:ins runs); we wrap them. Deletions are NOT
    in the markdown by default (Docling drops w:del); we re-insert them
    with [- ... -] markers, placing them after the text where they originally
    appeared if findable. Best-effort.
    """
    for rev in revisions:
        if not rev.text.strip():
            continue
        if rev.kind == "ins":
            # Wrap the inserted text in [+ ... +].
            target = rev.text
            if target in markdown:
                markdown = markdown.replace(target, f"[+ {target.strip()} +]", 1)
        elif rev.kind == "del":
            # Append the deletion marker — Docling has already dropped the
            # text, so we surface it next to the place it was deleted from.
            # Simplest reliable placement: append at end of markdown body
            # with a clear marker. (Refinement: locate by surrounding context
            # in a future iteration.)
            markdown = markdown.rstrip() + f"\n\n[- {rev.text.strip()} -]\n"
    return markdown


def _append_footnotes_section(markdown: str, footnotes: List[tuple[str, str]]) -> str:
    """Append a `## Footnotes` section at the end of the markdown. Replace
    Docling's inline footnote rendering (if any) with `[^N]` references.

    Docling's DOCX backend may render footnote text inline at the reference
    point or omit it entirely; we strip any duplicate inline rendering and
    rely on the appendix as the canonical location.
    """
    if not footnotes:
        return markdown

    # Append section
    lines = ["", "", "## Footnotes", ""]
    for fid, ftext in footnotes:
        lines.append(f"[^{fid}]: {ftext}")
    return markdown.rstrip() + "\n".join(lines) + "\n"


def _append_reviewer_comments(
    markdown: str, comments: List[_CommentEntry]
) -> str:
    """Append a `## Reviewer Comments` appendix. Best-effort: also insert
    `[C1]`, `[C2]` inline anchors after the referenced text if it's findable
    in the markdown. If not, the comment still appears in the appendix
    without an inline anchor.
    """
    if not comments:
        return markdown

    # Inline anchors (best-effort)
    for i, c in enumerate(comments, start=1):
        anchor_id = f"[C{i}]"
        if c.referenced_text and c.referenced_text in markdown:
            markdown = markdown.replace(
                c.referenced_text,
                f"{c.referenced_text} {anchor_id}",
                1,
            )

    # Appendix
    lines = ["", "", "## Reviewer Comments", ""]
    for i, c in enumerate(comments, start=1):
        date_str = f" ({c.date})" if c.date else ""
        ref_str = f"\n> Referenced text: \"{c.referenced_text}\"" if c.referenced_text else ""
        lines.append(f"[C{i}] **{c.author}**{date_str}: {c.text}{ref_str}")
        lines.append("")
    return markdown.rstrip() + "\n".join(lines) + "\n"


class DOCXConverter(BaseConverter):
    """Convert Word DOCX files to markdown via Docling + python-docx."""

    extensions = (".docx",)

    def convert(self, input_path: str, options: ConversionOptions) -> ConversionResult:
        logger = logging.getLogger("docx_converter")

        src = Path(input_path)
        if not src.exists():
            return ConversionResult(status="error", error=f"File not found: {input_path}")

        # 1. Open the DOCX with python-docx for auxiliary content.
        try:
            from docx import Document
        except ImportError:
            return ConversionResult(
                status="error",
                error=("python-docx is required for DOCX conversion. "
                       "Install with: pip install python-docx"),
            )

        try:
            doc = Document(str(src))
        except Exception as exc:
            return ConversionResult(
                status="error",
                error=f"Could not open DOCX: {exc}",
            )

        comments = _extract_comments(doc) if options.full else []
        footnotes = _extract_footnotes(doc)
        revisions = _extract_revisions(doc) if options.show_revisions else []

        # 2. Run Docling for prose / headings / tables.
        try:
            converter = DocumentConverter()
            docling_result = converter.convert(str(src))
            markdown = docling_result.document.export_to_markdown()
        except Exception as exc:
            logger.warning(f"Docling failed on {input_path}: {exc}")
            return ConversionResult(
                status="error",
                error=f"Docling could not parse the DOCX: {exc}",
            )

        # 3. Post-process.
        if options.show_revisions and revisions:
            markdown = _render_revisions_inline(markdown, revisions)

        if not options.keep_images:
            # Replace any inline image references with the placeholder.
            # Docling renders images as `![...](path)` — strip and replace.
            markdown = re.sub(r"!\[[^\]]*\]\([^)]*\)", "<!-- image -->", markdown)

        if footnotes:
            markdown = _append_footnotes_section(markdown, footnotes)

        if options.full and comments:
            markdown = _append_reviewer_comments(markdown, comments)

        if options.page_markers:
            markdown = _insert_section_markers(markdown)

        # 4. Cheap verifier.
        report = VerifyReport()
        report.results.append(check_non_empty(markdown))
        if report.overall == "FAIL":
            return ConversionResult(
                status="error",
                error="Cheap verification failed: output empty",
            )
        # Source word count: count words in the body via python-docx (rough but
        # consistent — same paragraphs Docling reads from).
        source_words = sum(count_words(p.text) for p in doc.paragraphs)
        output_words = count_words(markdown)
        report.results.append(
            check_word_retention(source_words, output_words, DOCX_WORD_RETENTION_MIN)
        )

        if report.overall == "WARN":
            markdown = f"<!-- VERIFY: WARN -->\n\n{markdown}"

        # 5. Write output.
        out_path = default_output_path(input_path, options.output_path)
        out_path.parent.mkdir(parents=True, exist_ok=True)
        out_path.write_text(markdown, encoding="utf-8")

        return ConversionResult(
            status="success",
            output_file=str(out_path),
            statistics={
                "words": output_words,
                "characters": len(markdown),
                "source_words": source_words,
                "verify_status": report.overall,
                "sections": markdown.count("<!-- Section "),
                "headings": len(_HEADING_RE.findall(markdown)),
                "comments": len(comments),
                "footnotes": len(footnotes),
                "revisions": len(revisions),
            },
        )
```
